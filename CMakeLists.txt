cmake_minimum_required(VERSION 3.16)
set(WKAL_BOARD "esp32s3-n4r2" CACHE STRING "ESP32-S3 memory target")
if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/targets/${WKAL_BOARD}.defaults")
    message(FATAL_ERROR "Unknown WKAL_BOARD: ${WKAL_BOARD}")
endif()
if(DEFINED WKAL_CONFIGURED_BOARD AND NOT WKAL_CONFIGURED_BOARD STREQUAL WKAL_BOARD)
    message(FATAL_ERROR "Build directory belongs to another board; use build/<target>")
endif()
set(WKAL_CONFIGURED_BOARD "${WKAL_BOARD}" CACHE INTERNAL "Board owning this build")
# Separate sdkconfig for each build directory.
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/targets/${WKAL_BOARD}.defaults")
set(IDF_TARGET "esp32s3")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(ps5_wkal_host)
